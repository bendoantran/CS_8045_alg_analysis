cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(ClosestPairComparison VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
# Using GCC/libstdc++ makes the Parallel STL backend explicit and comparable.
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Use GCC/g++ (Linux, WSL, or Homebrew GCC). See README.md.")
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
option(CP_FETCH_TBB "Build pinned oneTBB locally instead of requiring installation" OFF)
if(CP_FETCH_TBB)
  include(FetchContent)
  set(TBB_TEST OFF CACHE BOOL "" FORCE)
  set(TBB_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(TBB_STRICT OFF CACHE BOOL "" FORCE)
  set(TBBMALLOC_BUILD OFF CACHE BOOL "" FORCE)
  set(TBB_DISABLE_HWLOC_AUTOMATIC_SEARCH ON CACHE BOOL "" FORCE)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/oneTBB-v2021.13.0.tar.gz")
    set(CP_TBB_URL "${CMAKE_CURRENT_SOURCE_DIR}/oneTBB-v2021.13.0.tar.gz")
  else()
    set(CP_TBB_URL "https://codeload.github.com/oneapi-src/oneTBB/tar.gz/refs/tags/v2021.13.0")
  endif()
  FetchContent_Declare(oneTBB URL "${CP_TBB_URL}"
    URL_HASH SHA256=3ad5dd08954b39d113dc5b3f8a8dc6dc1fd5250032b7c491eb07aed5c94133e1)
  FetchContent_MakeAvailable(oneTBB)
else()
  find_package(TBB 2021 CONFIG REQUIRED COMPONENTS tbb)
endif()
add_library(closest_pair src/common.cpp src/sequential.cpp src/openmp.cpp src/tbb.cpp src/pstl.cpp)
target_include_directories(closest_pair PUBLIC include)
target_link_libraries(closest_pair PUBLIC OpenMP::OpenMP_CXX TBB::tbb)
# Explicitly disallow libstdc++'s silent serial PSTL fallback.
target_compile_definitions(closest_pair PUBLIC CP_REQUIRE_TBB_PSTL=1)
target_compile_options(closest_pair PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(closest_pair PUBLIC -ffp-contract=off)
foreach(algorithm IN ITEMS sequential openmp tbb pstl)
  add_executable(closest_${algorithm} src/main.cpp)
  target_compile_definitions(closest_${algorithm} PRIVATE CP_DEFAULT_ALGORITHM="${algorithm}")
  target_link_libraries(closest_${algorithm} PRIVATE closest_pair)
endforeach()
add_executable(benchmark src/main.cpp)
target_compile_definitions(benchmark PRIVATE CP_DEFAULT_ALGORITHM="all")
target_link_libraries(benchmark PRIVATE closest_pair)
include(CTest)
if(BUILD_TESTING)
  add_executable(correctness_tests tests/correctness.cpp)
  target_link_libraries(correctness_tests PRIVATE closest_pair)
  add_test(NAME correctness COMMAND correctness_tests)
  add_test(NAME cli_smoke COMMAND benchmark --points 250 --threads 2 --repeats 2 --warmups 1)
  add_test(NAME file_input COMMAND benchmark --input ${CMAKE_CURRENT_SOURCE_DIR}/data/example.csv --threads 2 --repeats 1 --warmups 0)
  add_test(NAME invalid_points COMMAND benchmark --points 1)
  set_tests_properties(invalid_points PROPERTIES WILL_FAIL TRUE)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME table_metrics COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/table_metrics.py)
    add_test(NAME benchmark_csv COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_csv.py $<TARGET_FILE:benchmark>)
  endif()
  add_test(NAME invalid_threads COMMAND benchmark --threads 0)
  set_tests_properties(invalid_threads PROPERTIES WILL_FAIL TRUE)
endif()
